DBMS MODEL LAB - EXPERIMENT 7
ANY, ALL, IN, EXISTS, NOT EXISTS, UNION AND INTERSECT

AIM:
To execute SQL queries using ANY, ALL, IN, EXISTS, NOT EXISTS, UNION and INTERSECT.

PROGRAM:
CREATE TABLE STUDENT(ROLLNO NUMBER(3), NAME VARCHAR2(20), MARKS NUMBER(3));
INSERT INTO STUDENT VALUES (101,'RAVI',85);
INSERT INTO STUDENT VALUES (102,'RAJU',75);
INSERT INTO STUDENT VALUES (103,'SITA',90);

CREATE TABLE STUDENT2(ROLLNO NUMBER(3), NAME VARCHAR2(20), MARKS NUMBER(3));
INSERT INTO STUDENT2 VALUES (101,'RAVI',85);
INSERT INTO STUDENT2 VALUES (104,'RAM',80);
INSERT INTO STUDENT2 VALUES (105,'KIRAN',70);

IN:
SELECT * FROM STUDENT WHERE MARKS IN (75,90);

ANY:
SELECT * FROM STUDENT
WHERE MARKS > ANY (SELECT MARKS FROM STUDENT2);

ALL:
SELECT * FROM STUDENT
WHERE MARKS > ALL (SELECT MARKS FROM STUDENT2);

EXISTS:
SELECT * FROM STUDENT S
WHERE EXISTS (SELECT * FROM STUDENT2 S2 WHERE S.ROLLNO=S2.ROLLNO);

NOT EXISTS:
SELECT * FROM STUDENT S
WHERE NOT EXISTS (SELECT * FROM STUDENT2 S2 WHERE S.ROLLNO=S2.ROLLNO);

UNION:
SELECT NAME FROM STUDENT UNION SELECT NAME FROM STUDENT2;

INTERSECT:
SELECT NAME FROM STUDENT INTERSECT SELECT NAME FROM STUDENT2;

RESULT:
All specified operators were executed successfully.
