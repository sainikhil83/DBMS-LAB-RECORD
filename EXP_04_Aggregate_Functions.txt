DBMS MODEL LAB - EXPERIMENT 4
AGGREGATE FUNCTIONS, GROUP BY AND HAVING

AIM:
To perform aggregate functions and queries using GROUP BY and HAVING.

SQL:
SELECT COUNT(*) FROM STUDENT;
SELECT SUM(MARKS) FROM STUDENT;
SELECT AVG(MARKS) FROM STUDENT;
SELECT MAX(MARKS) FROM STUDENT;
SELECT MIN(MARKS) FROM STUDENT;

ALTER TABLE STUDENT ADD DEPT VARCHAR2(10);

UPDATE STUDENT SET DEPT='CSE' WHERE ROLLNO=101;
UPDATE STUDENT SET DEPT='CSE' WHERE ROLLNO=102;
UPDATE STUDENT SET DEPT='ECE' WHERE ROLLNO=103;

SELECT DEPT, COUNT(*) FROM STUDENT GROUP BY DEPT;

SELECT DEPT, COUNT(*)
FROM STUDENT
GROUP BY DEPT
HAVING COUNT(*) > 1;

RESULT:
Aggregate functions, GROUP BY and HAVING were executed successfully.
