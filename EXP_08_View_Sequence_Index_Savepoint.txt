DBMS MODEL LAB - EXPERIMENT 8
VIEW, SEQUENCE, INDEX AND SAVEPOINT

AIM:
To create and execute View, Sequence, Index and Savepoint.

PROGRAM:
CREATE TABLE STUDENT(ROLLNO NUMBER(3), NAME VARCHAR2(20), MARKS NUMBER(3));

INSERT INTO STUDENT VALUES (101,'RAVI',85);
INSERT INTO STUDENT VALUES (102,'RAJU',75);
INSERT INTO STUDENT VALUES (103,'SITA',90);

VIEW:
CREATE VIEW STUDENT_VIEW AS
SELECT ROLLNO, NAME, MARKS FROM STUDENT;
SELECT * FROM STUDENT_VIEW;

SEQUENCE:
CREATE SEQUENCE STUDENT_SEQ START WITH 104 INCREMENT BY 1;
SELECT STUDENT_SEQ.NEXTVAL FROM DUAL;

INDEX:
CREATE INDEX STUDENT_INDEX ON STUDENT(NAME);

SAVEPOINT:
UPDATE STUDENT SET MARKS=95 WHERE ROLLNO=101;
SAVEPOINT S1;
UPDATE STUDENT SET MARKS=70 WHERE ROLLNO=102;
ROLLBACK TO S1;
SELECT * FROM STUDENT;

RESULT:
View, Sequence, Index and Savepoint were created and executed successfully.
