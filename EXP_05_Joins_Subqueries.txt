DBMS MODEL LAB - EXPERIMENT 5
RELATIONSHIPS, JOINS AND SUBQUERIES

AIM:
To create relationships between tables and perform JOIN and SUBQUERY operations.

SQL:
CREATE TABLE STUDENT
(
 SID NUMBER(3) PRIMARY KEY,
 NAME VARCHAR2(20),
 CID NUMBER(3)
);

INSERT INTO STUDENT VALUES (101,'RAVI',1);
INSERT INTO STUDENT VALUES (102,'RAJU',2);
INSERT INTO STUDENT VALUES (103,'SITA',1);

CREATE TABLE COURSE
(
 CID NUMBER(3) PRIMARY KEY,
 CNAME VARCHAR2(20)
);

INSERT INTO COURSE VALUES (1,'CSE');
INSERT INTO COURSE VALUES (2,'ECE');
INSERT INTO COURSE VALUES (3,'AI');

INNER JOIN:
SELECT STUDENT.NAME, COURSE.CNAME
FROM STUDENT
INNER JOIN COURSE ON STUDENT.CID=COURSE.CID;

LEFT JOIN:
SELECT STUDENT.NAME, COURSE.CNAME
FROM STUDENT
LEFT JOIN COURSE ON STUDENT.CID=COURSE.CID;

SUBQUERY:
SELECT NAME FROM STUDENT
WHERE CID=(SELECT CID FROM COURSE WHERE CNAME='CSE');

RESULT:
JOIN and subquery operations were executed successfully.
