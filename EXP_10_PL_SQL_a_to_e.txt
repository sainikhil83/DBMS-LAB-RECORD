10(a) CONDITION
SET SERVEROUTPUT ON;
DECLARE
 N NUMBER;
BEGIN
 N := &N;
 IF N > 0 THEN
   DBMS_OUTPUT.PUT_LINE('Positive Number');
 ELSE
   DBMS_OUTPUT.PUT_LINE('Negative Number');
 END IF;
END;
/
OUTPUT for 10:
Positive Number

10(b) EXCEPTION HANDLING
SET SERVEROUTPUT ON;
DECLARE
 A NUMBER := 10;
 B NUMBER := 0;
 C NUMBER;
BEGIN
 C := A / B;
 DBMS_OUTPUT.PUT_LINE(C);
EXCEPTION
 WHEN ZERO_DIVIDE THEN
   DBMS_OUTPUT.PUT_LINE('Cannot divide by zero');
END;
/
OUTPUT:
Cannot divide by zero

10(c) PROCEDURE

CREATE OR REPLACE PROCEDURE GREET

 DBMS_OUTPUT.PUT_LINE('Hello Student');

EXEC GREET;

FUNCTION
CREATE OR REPLACE FUNCTION ADD_NUM
RETURN NUMBER
AS

 RETURN 10+20;

OUTPUT:
30

10(d) TRIGGERS
CREATE TABLE STUDENT(ROLLNO NUMBER(3), NAME VARCHAR2(20), MARKS NUMBER(3));

INSERT INTO STUDENT VALUES (101,'RAVI',85);
INSERT INTO STUDENT VALUES (102,'RAJU',75);
INSERT INTO STUDENT VALUES (103,'SITA',90);

CREATE OR REPLACE TRIGGER STUDENT_INSERT
AFTER INSERT ON STUDENT
BEGIN
 DBMS_OUTPUT.PUT_LINE('Record Inserted');
END;
/

CREATE OR REPLACE TRIGGER STUDENT_UPDATE
AFTER UPDATE ON STUDENT
BEGIN
 DBMS_OUTPUT.PUT_LINE('Record Updated');
END;
/

CREATE OR REPLACE TRIGGER STUDENT_DELETE
AFTER DELETE ON STUDENT
BEGIN
 DBMS_OUTPUT.PUT_LINE('Record Deleted');
END;
/

10(e) CURSORS
IMPLICIT:
SET SERVEROUTPUT ON;
BEGIN
 UPDATE STUDENT SET MARKS=90 WHERE ROLLNO=101;
 DBMS_OUTPUT.PUT_LINE(SQL%ROWCOUNT || ' Record Updated');
END;
/

EXPLICIT:
SET SERVEROUTPUT ON;
DECLARE
 CURSOR C1 IS SELECT NAME, MARKS FROM STUDENT;
 V_NAME STUDENT.NAME%TYPE;
 V_MARKS STUDENT.MARKS%TYPE;
BEGIN
 OPEN C1;
 LOOP
  FETCH C1 INTO V_NAME,V_MARKS;
  EXIT WHEN C1%NOTFOUND;
  DBMS_OUTPUT.PUT_LINE(V_NAME || ' ' || V_MARKS);
 END LOOP;
 CLOSE C1;
END;
/

RESULT:
PL/SQL condition, exception, procedure, function, triggers and cursors were executed successfully.
